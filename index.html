const categories = ['Exam', 'Study', 'Assignment', 'Personal'];
const json = (data, status = 200) => Response.json(data, {status, headers: {'Cache-Control':'no-store'}});

export default {
  async fetch(request, env) {
    const url = new URL(request.url);
    const accountId = request.headers.get('oai-authenticated-user-id');
    const email = request.headers.get('oai-authenticated-user-email')?.trim().toLowerCase() || '';
    const owner = Boolean(email && email === env.OWNER_EMAIL);
    if (!accountId) {
      return new Response('Sign in to open ExamDesk.', {status:401, headers:{'Cache-Control':'no-store'}});
    }
    if (url.pathname === '/') {
      return new Response(owner ? page : page.replace(ownerButton, ''), {headers:{
        'Content-Type':'text/html; charset=utf-8', 'Cache-Control':'no-store',
        'Content-Security-Policy':"default-src 'self'; script-src 'unsafe-inline'; style-src 'unsafe-inline'; img-src 'self' data:; base-uri 'none'; form-action 'self'",
        'X-Content-Type-Options':'nosniff'
      }});
    }
    if (url.pathname === '/api/owner/events') {
      if (!owner) return json({error:'Page not found'}, 404);
      if (request.method !== 'GET') return json({error:'Method not supported.'}, 405);
      try {
        const {results} = await env.DB.prepare(
          'SELECT title, kind, day, start, end, location, notes, account_id, account_email FROM events WHERE account_id <> ? ORDER BY account_email, account_id, day, start, title'
        ).bind(accountId).all();
        return json(results);
      } catch (error) {
        console.error('Owner timetable query failed', error);
        return json({error:'Could not load timetables.'}, 503);
      }
    }
    const exporting = url.pathname === '/calendar.ics';
    if (!exporting && url.pathname !== '/api/events') return json({error:'Page not found'}, 404);
    const ownEvents = '(account_id = ? OR (? = 1 AND account_id IS NULL))';
    const scope = [accountId, Number(owner)];
    if (exporting && request.method !== 'GET') return json({error:'Method not supported.'}, 405);
    try {
      if (request.method === 'GET') {
        const {results} = await env.DB.prepare(
          'SELECT id, title, kind, day, start, end, location, notes FROM events WHERE ' + ownEvents + ' ORDER BY day, start, title'
        ).bind(...scope).all();
        if (exporting) {
          return new Response(calendarFile(results), {headers:{
            'Content-Type':'text/calendar; charset=utf-8',
            'Content-Disposition':'attachment; filename="examdesk-timetable.ics"',
            'Cache-Control':'private, no-store',
            'X-Content-Type-Options':'nosniff'
          }});
        }
        return json(results);
      }
      if (request.headers.get('Origin') !== url.origin) return json({error:'Request origin was rejected.'}, 403);
      if (!['POST','DELETE'].includes(request.method)) return json({error:'Method not supported.'}, 405);
      const raw = await request.text();
      if (raw.length > 8000) return json({error:'Event is too large.'}, 413);
      let input;
      try { input = JSON.parse(raw); } catch { return json({error:'Invalid event data.'}, 400); }
      if (!input || typeof input !== 'object') return json({error:'Invalid event data.'}, 400);
      const {id,title,kind,day,start,end,location,notes} = input;
      if (id && !/^[0-9a-f-]{36}$/.test(id)) return json({error:'Invalid event ID.'}, 400);
      if (request.method === 'DELETE') {
        if (!id) return json({error:'Missing event ID.'}, 400);
        const result = await env.DB.prepare('DELETE FROM events WHERE id=? AND ' + ownEvents).bind(id,...scope).run();
        if (!result.meta.changes) return json({error:'Event not found.'},404);
        return json({ok:true});
      }
      const limit = (value, max) => typeof value === 'string' && value.length <= max;
      const parsed = new Date(day + 'T12:00:00Z');
      const validDay = /^\d{4}-\d{2}-\d{2}$/.test(day) && Number.isFinite(+parsed) && parsed.toISOString().slice(0,10) === day;
      const time = /^([01]\d|2[0-3]):[0-5]\d$/;
      if (!limit(title,120) || !title.trim() || !categories.includes(kind) || !validDay ||
          !(start === '' || time.test(start)) || !(end === '' || (time.test(end) && start && end > start)) ||
          !limit(location,150) || !limit(notes,1500)) return json({error:'Enter a title, valid date and an end time after the start.'}, 400);
      const data = [title.trim(),kind,day,start,end,location.trim(),notes.trim()];
      if (id) {
        const result = await env.DB.prepare('UPDATE events SET title=?, kind=?, day=?, start=?, end=?, location=?, notes=?, account_id=?, account_email=? WHERE id=? AND ' + ownEvents).bind(...data,accountId,email,id,...scope).run();
        if (!result.meta.changes) return json({error:'This event has already been removed.'},404);
      } else {
        await env.DB.prepare('INSERT INTO events (title,kind,day,start,end,location,notes,id,account_id,account_email) VALUES (?,?,?,?,?,?,?,?,?,?)').bind(...data,crypto.randomUUID(),accountId,email).run();
      }
      return json({ok:true});
    } catch (error) {
      console.error('Calendar storage failed', error);
      return json({error:'Calendar storage is temporarily unavailable.'}, 503);
    }
  }
};

function calendarFile(list) {
  const escapeIcs = value => value.replace(/\\/g,'\\\\').replace(/\r?\n/g,'\\n').replace(/([,;])/g,'\\$1');
  const stamp = new Date().toISOString().replace(/[-:]/g,'').replace(/\.\d+Z$/,'Z');
  const lines = ['BEGIN:VCALENDAR','VERSION:2.0','PRODID:-//ExamDesk//Timetable//EN','CALSCALE:GREGORIAN'];
  for (const e of list) {
    const day = e.day.replaceAll('-','');
    const timed = time => new Date(e.day+'T'+time+':00+05:30').toISOString().replace(/[-:]/g,'').replace(/\.\d+Z$/,'Z');
    lines.push('BEGIN:VEVENT','UID:'+e.id+'@examdesk','DTSTAMP:'+stamp,'SUMMARY:'+escapeIcs(e.title),'CATEGORIES:'+e.kind,
      e.start ? 'DTSTART:'+timed(e.start) : 'DTSTART;VALUE=DATE:'+day);
    if (e.end) lines.push('DTEND:'+timed(e.end));
    if (e.location) lines.push('LOCATION:'+escapeIcs(e.location));
    if (e.notes) lines.push('DESCRIPTION:'+escapeIcs(e.notes));
    lines.push('END:VEVENT');
  }
  lines.push('END:VCALENDAR','');
  const byteLength = char => new TextEncoder().encode(char).length;
  const folded = lines.map(line => {
    let output = '', count = 0;
    for (const char of line) {
      const size = byteLength(char);
      if (count + size > 75) { output += '\r\n '; count = 1; }
      output += char;
      count += size;
    }
    return output;
  });
  return folded.join('\r\n');
}

      cells += '<div class="cell'+(date.getMonth()!==anchor.getMonth()&&view==='month'?' out':'')+(day===selected?' selected':'')+'" data-cell="'+day+'"><button class="date'+(day===today()?' today':'')+'" data-select="'+day+'" aria-label="'+format(date,{dateStyle:'full'})+'" aria-pressed="'+(day===selected)+'">'+date.getDate()+'</button>'+chips+more+'</div>';
    }
    $('calendar').innerHTML = '<div class="weekdays">'+weekdays+'</div><div class="grid '+view+'">'+cells+'</div>';
  }

  $('upcoming').innerHTML = events.filter(e => e.kind === 'Exam' && e.day >= today()).slice(0,5).map(e => entry(e,true)).join('') || empty('No upcoming exams.');
}

  const result = await fetch('/api/events',{method,headers:body?{'Content-Type':'application/json'}:{},body:body?JSON.stringify(body):undefined});
  if (result.redirected) throw new Error('Sign in again and reload to update the timetable.');
  const data = await result.json();
  if (!result.ok) throw new Error(data.error || 'The calendar could not be updated.');

async function mutate(method, data) {
  if (busy) return;
  busy = true;
  $('save').disabled = $('remove').disabled = true;
  $('formError').textContent = '';
  try {
    await api(method,data);
    if (method !== 'DELETE') {
      selected = data.day;
      anchor = toDate(selected);
    }
    $('editor').close();
    notice(method === 'DELETE' ? 'Event deleted.' : 'Event saved.');
    try { await load(); } catch { notice('The change was saved. Reload to refresh the timetable.',true); }
  } catch (error) {
    $('formError').textContent = error.message;
  } finally {
    busy = false;
    $('save').disabled = $('remove').disabled = false;
  }
}
